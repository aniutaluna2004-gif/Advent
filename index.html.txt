<script>
const tg = window.Telegram.WebApp;
tg.expand(); // Разворачиваем окно на весь экран

// ВСТАВЬТЕ СЮДА ВАШ WEBHOOK ИЗ N8N
const N8N_WEBHOOK_URL = 'https://n8n-yulia-yulialunan8n.waw0.amvera.tech/webhook/28b6c2b0-a1f6-484c-8e43-b009fefc7c8c';

const calendar = document.getElementById('calendar');
const today = new Date();
// Для теста ставим 10 число, в декабре можно использовать today.getDate()
const currentDay = today.getDate(); 

for (let day = 1; day <= 31; day++) {
const btn = document.createElement('button');
btn.className = 'day-btn';
btn.innerText = day <= currentDay ? 🎁 ${day} : 🔒 ${day};

if (day > currentDay) {
btn.classList.add('locked');
} else {
btn.onclick = () => openGift(day);
}
calendar.appendChild(btn);
}

function openGift(day) {
// 1. Запускаем конфетти
confetti({
particleCount: 100,
spread: 70,
origin: { y: 0.6 }
});

// 2. Вибрация телефона (приятный отклик)
tg.HapticFeedback.notificationOccurred('success');

// 3. Отправляем данные в n8n, чтобы бот прислал задание и PDF в чат
const userId = tg.initDataUnsafe?.user?.id;

fetch(N8N_WEBHOOK_URL, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify({ chat_id: userId, day: day })
}).then(() => {
tg.showAlert(Подарок за ${day} декабря отправлен вам в чат с ботом!);
}).catch(() => {
tg.showAlert(День ${day} открыт! Проверьте чат с ботом.);
});
}
